const agencyProjects = pgTable(
  "agency_projects",
  {
    id: uuid("id").primaryKey(),
    // Historical proposal: <=255 UTF-16 units. SQL length constraint UNDECIDED.
    title: text("title").notNull(),
    ownerId: uuid("owner_id").notNull().references(() => users.id, { onDelete: "restrict" }),
    dealId: uuid("deal_id").references(() => deals.id, { onDelete: "restrict" }),
    personId: uuid("person_id").references(() => persons.id, { onDelete: "restrict" }),
    orgId: uuid("org_id").references(() => organizations.id, { onDelete: "restrict" }),
    state: text("state").notNull(),
    revision: bigint("revision", { mode: "number" }).notNull().default(0),
    archivedFrom: text("archived_from"),
    createdAt: timestamp("created_at", { withTimezone: true }).notNull(),
    updatedAt: timestamp("updated_at", { withTimezone: true }).notNull(),
  },
  (t) => [
    check("agency_projects_revision_ck", sql`${t.revision} BETWEEN 0 AND 9007199254740991`),
    check("agency_projects_state_ck", sql`${t.state} IN
      ('planned', 'active', 'on-hold', 'completed', 'canceled', 'archived')`),
    check("agency_projects_archive_ck", sql`
      (${t.state} = 'archived' AND ${t.archivedFrom} IS NOT NULL AND
        ${t.archivedFrom} IN ('planned', 'active', 'on-hold', 'completed', 'canceled')) OR
      (${t.state} <> 'archived' AND ${t.archivedFrom} IS NULL)`),
    check("agency_projects_anchor_ck", sql`
      ${t.dealId} IS NOT NULL OR ${t.personId} IS NOT NULL OR ${t.orgId} IS NOT NULL`),
    index("agency_projects_owner_state_id_idx").on(t.ownerId, t.state, t.id),
    index("agency_projects_deal_idx").on(t.dealId),
    index("agency_projects_person_idx").on(t.personId),
    index("agency_projects_org_idx").on(t.orgId),
  ],
);
// No unique customer/deal constraint. A deliberate new project may reuse a CRM link.
